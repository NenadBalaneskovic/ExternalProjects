# -*- coding: utf-8 -*-

"""
thread_safe_fifo_pipeline.py

A thread‑safe FIFO pipeline that accepts sparse‑matrix problems
(A, b, λ) and returns the Tikhonov‑regularised solution in real time.
"""

from __future__ import annotations

import threading
import queue
import itertools
import logging
from typing import Callable, Iterable, Tuple, Optional, Any

import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla

    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(threadName)s – %(message)s",
    datefmt="%H:%M:%S",
)

log = logging.getLogger(__name__)

# ----------------------------------------------------------------------
# Helper: diagonal Tikhonov solver for sparse matrices
# ----------------------------------------------------------------------
def solve_diagonal_tikhonov(
    A: sp.spmatrix,
    b: np.ndarray,
    lam: float,
    *,
    method: str = "direct",
    maxiter: int = 1000,
    tol: float = 1e-8,
) -> np.ndarray:
    """
    Solve (A.T @ A + λ I) x = A.T @ b for a sparse A.

    Parameters
    ----------
    A : sp.spmatrix
        The (m × n) sparse matrix.
    b : np.ndarray
        Right‑hand side of size (m,).  Must be dense because most solvers
        expect a dense vector.
    lam : float
        Regularisation parameter λ ≥ 0.
    method : {'direct', 'cg'}, optional
        * ``'direct'`` – use a sparse direct solver (SuperLU).  Fast for
          medium‑size problems (<~10⁴ unknowns) and robust.
        * ``'cg'`` – use Conjugate Gradient on the normal equations.
          Only works when the matrix is symmetric positive definite,
          which is guaranteed after the Tikhonov term.
    maxiter, tol : passed to the iterative solver.

    Returns
    -------
    x : np.ndarray
        The regularised solution (shape (n,)).
    """
    if not sp.isspmatrix_csr(A) and not sp.isspmatrix_csc(A):
        # CSR/CSC are the only formats that support efficient matmat / transpose.
        A = A.tocsr()

    # Build the normal‑equation matrix *once*.
    ATA = A.T @ A                     # sparse (n × n)
    if lam != 0.0:
        # Add λ on the diagonal – the most efficient way is to use sp.eye.
        ATA = ATA + lam * sp.eye(ATA.shape[0], format="csr")

    ATb = A.T @ b                     # dense (n,)

    if method == "direct":
        # SuperLU (spsolve) works on any square sparse matrix.
        x = spla.spsolve(ATA, ATb)
    elif method == "cg":
        # CG works because ATA+λI is SPD.
        x, info = spla.cg(ATA, ATb, maxiter=maxiter, tol=tol)
        if info != 0:
            raise RuntimeError(f"CG did not converge (info={info})")
    else:
        raise ValueError(f"Unsupported method {method!r}")

    return x


# ----------------------------------------------------------------------
# Pipeline class
# ----------------------------------------------------------------------
class SparseTikhonovPipeline:
    """
    A thread‑safe FIFO pipeline that processes sparse‑matrix Tikhonov problems.

    Each item put into the pipeline must be a tuple ``(A, b, λ)`` where

    * ``A`` is a scipy.sparse matrix (any format, will be converted to CSR/CSC),
    * ``b`` is a 1‑D numpy array of length ``A.shape[0]``,
    * ``λ`` is a non‑negative float (regularisation parameter).

    The pipeline can run with a configurable number of consumer threads.
    Results are emitted either via a user supplied callback or placed in an
    optional ``result_queue`` that preserves the same FIFO order.
    """

    # ------------------------------------------------------------------
    # Construction / configuration
    # ------------------------------------------------------------------
    def __init__(
        self,
        num_workers: int = 1,
        *,
        solve_method: str = "direct",
        maxiter: int = 1000,
        cg_tol: float = 1e-8,
        result_queue: Optional[queue.Queue] = None,
        result_callback: Optional[Callable[[int, np.ndarray], Any]] = None,
        name: str = "SparseTikhonovPipeline",
    ) -> None:
        """
        Parameters
        ----------
        num_workers : int, default 1
            Number of consumer threads that will pull items from the input queue.
        solve_method : {'direct', 'cg'}, default 'direct'
            Which algorithm to use inside :func:`solve_diagonal_tikhonov`.
        maxiter, cg_tol : passed to the CG solver if it is chosen.
        result_queue : queue.Queue, optional
            If supplied, each computed solution ``x`` is put into this queue
            together with a monotonically increasing *job id*.
        result_callback : Callable[[int, np.ndarray], Any], optional
            Alternative to ``result_queue`` – called for each finished job.
            The first argument is the job id, the second the solution vector.
        name : str
            Human‑readable name used for the internal threads (helpful when
            debugging).
        """
        if num_workers < 1:
            raise ValueError("num_workers must be >= 1")

        self._input_q: queue.Queue[Tuple[int, Tuple[sp.spmatrix, np.ndarray, float]]] = queue.Queue()
        self._result_q = result_queue
        self._callback = result_callback
        self._solve_method = solve_method
        self._maxiter = maxiter
        self._cg_tol = cg_tol
        self._workers: list[threading.Thread] = []
        self._stop_event = threading.Event()
        self._job_counter = itertools.count()          # unique, monotonic ids

        # ------------------------------------------------------------------
        # Start the consumer threads
        # ------------------------------------------------------------------
        for i in range(num_workers):
            t = threading.Thread(
                target=self._worker,
                name=f"{name}-worker-{i + 1}",
                daemon=True,
            )
            t.start()
            self._workers.append(t)

        log.info("Pipeline created with %d worker thread(s).", num_workers)

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------
    def put(self, A: sp.spmatrix, b: np.ndarray, lam: float) -> int:
        """
        Enqueue a new problem.

        Returns
        -------
        job_id : int
            The identifier that will be associated with the solution when it
            appears in the result queue / callback.
        """
        if not sp.issparse(A):
            raise TypeError("A must be a scipy.sparse matrix")
        if b.ndim != 1:
            raise ValueError("b must be a 1‑D array")
        if A.shape[0] != b.shape[0]:
            raise ValueError("Incompatible dimensions: A.shape[0] != len(b)")
        if lam < 0:
            raise ValueError("Regularisation parameter λ must be non‑negative")

        job_id = next(self._job_counter)
        # Store the raw tuple; conversion to CSR/CSC is deferred to the worker.
        self._input_q.put((job_id, (A, b, lam)))
        log.debug("Enqueued job %d (λ=%.3g)", job_id, lam)
        return job_id

    def join(self, timeout: Optional[float] = None) -> None:
        """
        Block until all queued items have been processed.

        Parameters
        ----------
        timeout : float or None
            Passed to ``queue.Queue.join`` – ``None`` means wait forever.
        """
        self._input_q.join()
        log.info("All queued jobs have been processed.")

    def stop(self, graceful: bool = True, timeout: Optional[float] = None) -> None:
        """
        Signal the workers to terminate.

        Parameters
        ----------
        graceful : bool, default True
            If True, the method first drains the input queue (i.e. works like
            ``join``) and then stops the threads.  If False, the workers are
            asked to stop immediately – any pending jobs will be lost.
        timeout : float or None
            Maximum time to wait for the worker threads to finish.
        """
        if graceful:
            self.join()
        self._stop_event.set()

        # Insert sentinel values so that each thread wakes up.
        for _ in self._workers:
            self._input_q.put((None, None))

        for t in self._workers:
            t.join(timeout=timeout)
        log.info("Pipeline stopped (graceful=%s).", graceful)

    # ------------------------------------------------------------------
    # Internal worker implementation
    # ------------------------------------------------------------------
    def _worker(self) -> None:
        """
        Consumer thread: pull jobs from the input queue, solve them and push
        the results out.
        """
        while not self._stop_event.is_set():
            job_id, payload = self._input_q.get()
            try:
                # Sentinel used for shutdown
                if job_id is None and payload is None:
                    self._input_q.task_done()
                    break

                A, b, lam = payload
                # Ensure efficient format for the linear algebra
                if not sp.isspmatrix_csr(A) and not sp.isspmatrix_csc(A):
                    A = A.tocsr()

                # ----------------------------------------------------------
                # Solve the regularised system
                # ----------------------------------------------------------
                try:
                    x = solve_diagonal_tikhonov(
                        A,
                        b,
                        lam,
                        method=self._solve_method,
                        maxiter=self._maxiter,
                        tol=self._cg_tol,
                    )
                except Exception as exc:
                    log.exception("Failed to solve job %d", job_id)
                    # Propagate the exception downstream (user decides what to do)
                    x = exc

                # ----------------------------------------------------------
                # Emit result
                # ----------------------------------------------------------
                if self._callback is not None:
                    try:
                        self._callback(job_id, x)
                    except Exception:
                        log.exception("User callback raised for job %d", job_id)
                elif self._result_q is not None:
                    self._result_q.put((job_id, x))

                log.debug("Job %d completed (λ=%.3g).", job_id, lam)
            finally:
                # Always mark the task as done, even if an exception happened.
                self._input_q.task_done()
