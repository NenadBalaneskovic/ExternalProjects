import joblib

def batch_train_evaluate(train_data, test_data, model, batch_size, model_type="ML", aggregation_method="auto"):
    """
    Generalized wrapper function for batch-wise training and evaluation of large datasets.

    Parameters:
    - train_data: Path to CSV or DataFrame (large time-series dataset)
    - test_data: Path to CSV or DataFrame (test dataset)
    - model: ML/DL model object
    - batch_size: Number of rows per chunk
    - model_type: "ML" (Incremental ML), "DL" (Deep Learning)
    - aggregation_method: "incremental", "moving_average", "ensemble", "auto"

    Returns:
    - Optimal parameters aggregated across all chunks
    """

    # Determine chunking strategy (Pandas, Dask, Numpy memmap)
    if isinstance(train_data, str):
        chunk_iter = pd.read_csv(train_data, chunksize=batch_size)
    elif isinstance(train_data, pd.DataFrame):
        chunk_iter = np.array_split(train_data, len(train_data) // batch_size)
    elif isinstance(train_data, dd.DataFrame):
        chunk_iter = train_data.to_delayed()
    else:
        raise ValueError("Unsupported train_data format.")

    # Store chunk results
    param_list = []
    aggregated_params = None

    # Training loop across chunks
    for chunk in chunk_iter:
        X_chunk = chunk.iloc[:, :-1]  # Assume last column is target
        y_chunk = chunk.iloc[:, -1]

        # Model training: Incremental learning, gradient accumulation, rolling updates
        if model_type == "ML" and hasattr(model, "partial_fit"):  
            model.partial_fit(X_chunk, y_chunk)
        elif model_type == "DL":  
            model.train_on_batch(X_chunk, y_chunk)  # Typical DL batch training
        else:
            model.fit(X_chunk, y_chunk)  # Default full batch training

        # Collect parameters
        params = model.get_params() if hasattr(model, "get_params") else model.state_dict()
        param_list.append(params)

    # Aggregation method selection
    if aggregation_method == "incremental" or (aggregation_method == "auto" and model_type == "ML"):
        aggregated_params = param_list[-1]  # Last trained state for incremental learning
    elif aggregation_method == "moving_average" or (aggregation_method == "auto" and model_type == "DL"):
        aggregated_params = {key: np.mean([p[key] for p in param_list]) for key in param_list[0].keys()}
    elif aggregation_method == "ensemble":
        aggregated_params = np.mean(param_list)  # Ensemble averaging (can be refined per case)

    # Apply final parameters to model
    if hasattr(model, "set_params"):
        model.set_params(**aggregated_params)
    elif hasattr(model, "load_state_dict"):
        model.load_state_dict(aggregated_params)

    return aggregated_params

def custom_loss(y_true, y_pred, loss_type="mse"):
    """
    Implements custom loss functions based on user preference.
    """
    if loss_type == "mse":
        return np.mean((y_true - y_pred) ** 2)  # Mean Squared Error
    elif loss_type == "mae":
        return np.mean(np.abs(y_true - y_pred))  # Mean Absolute Error
    elif loss_type == "huber":
        delta = 1.0
        residual = np.abs(y_true - y_pred)
        return np.where(residual <= delta, 0.5 * residual ** 2, delta * (residual - 0.5 * delta)).mean()
    elif loss_type == "quantile":
        quantile = 0.5  # Adjust this as needed
        return np.mean(np.maximum(quantile * (y_true - y_pred), (1 - quantile) * (y_pred - y_true)))
    elif loss_type == "mase":
        scale = np.mean(np.abs(np.diff(y_true)))  # Scale factor for MASE
        return np.mean(np.abs(y_true - y_pred) / scale)
    else:
        raise ValueError("Unsupported loss function type.")

def time_series_cross_validation(model, X, y, cv_type="expanding", n_splits=5):
    """
    Implements time-series-specific cross-validation strategies.
    """
    if cv_type == "expanding":
        cv = TimeSeriesSplit(n_splits=n_splits)
    elif cv_type == "sliding":
        cv = TimeSeriesSplit(n_splits=n_splits, max_train_size=None)
    else:
        raise ValueError("Invalid time-series CV type.")

    scores = []
    for train_idx, test_idx in cv.split(X):
        X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
        y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]

        model.fit(X_train, y_train)
        y_pred = model.predict(X_test)
        loss = custom_loss(y_test, y_pred, loss_type="mase")  # Example loss function

        scores.append(loss)

    return np.mean(scores)  # Average CV score

def create_time_series_features(df, lags=3, rolling_window=5):
    """
    Generates lagged variables, rolling statistics, and datetime features for time-series.
    """
    df = df.copy()

    # Lagged features
    for lag in range(1, lags + 1):
        df[f'lag_{lag}'] = df.iloc[:, -1].shift(lag)

    # Rolling statistics
    df['rolling_mean'] = df.iloc[:, -1].rolling(window=rolling_window).mean()
    df['rolling_std'] = df.iloc[:, -1].rolling(window=rolling_window).std()

    # Datetime features (if applicable)
    if isinstance(df.index, pd.DatetimeIndex):
        df['day_of_week'] = df.index.dayofweek
        df['month'] = df.index.month
        df['hour'] = df.index.hour

    return df.fillna(0)

    delta = df[column].diff(1)
    gain = np.where(delta > 0, delta, 0)
    loss = np.where(delta < 0, -delta, 0)

    avg_gain = pd.Series(gain).rolling(window=window, min_periods=1).mean()
    avg_loss = pd.Series(loss).rolling(window=window, min_periods=1).mean()
    
    rs = avg_gain / avg_loss
    return 100 - (100 / (1 + rs))

    if df["timestamp"].dtype != "datetime64[ns]":

    return df

### **2. Feature Engineering (RSI, Bollinger Bands, Rolling Stats)**
def create_time_series_features(df, lags=3, rolling_window=5):
    df = df.copy()
    df = handle_missing_timestamps(df)

    # Lagged features
    for lag in range(1, lags + 1):
        df[f'lag_{lag}'] = df.iloc[:, -1].shift(lag)

    # Rolling statistics
    df['rolling_mean'] = df.iloc[:, -1].rolling(window=rolling_window).mean()
    df['rolling_std'] = df.iloc[:, -1].rolling(window=rolling_window).std()

    # RSI
    df["RSI"] = calculate_rsi(df)

    # Bollinger Bands
    bb = calculate_bollinger_bands(df)
    df = df.join(bb)

    return df.fillna(0)

def calculate_rsi(df, column="label", window=14):
    delta = df[column].diff(1)
    gain = np.where(delta > 0, delta, 0)
    loss = np.where(delta < 0, -delta, 0)

    avg_gain = pd.Series(gain).rolling(window=window, min_periods=1).mean()
    avg_loss = pd.Series(loss).rolling(window=window, min_periods=1).mean()
    
    rs = avg_gain / avg_loss
    return 100 - (100 / (1 + rs))

def calculate_bollinger_bands(df, column="label", window=20, std_dev=2):
    rolling_mean = df[column].rolling(window=window).mean()
    rolling_std = df[column].rolling(window=window).std()

    return pd.DataFrame({"BB_upper": rolling_mean + (std_dev * rolling_std), "BB_middle": rolling_mean, "BB_lower": rolling_mean - (std_dev * rolling_std)})

    sarimax_preds = sarimax_results.forecast(steps=len(X_test))

        "RandomForest": RandomForestRegressor(n_estimators=100)
    }

    ml_preds = {}
    for name, model in ml_models.items():
        model.fit(X_train, y_train)
        ml_preds[name] = model.predict(X_test)

    X_train_seq = np.expand_dims(X_train, axis=2)
    X_test_seq = np.expand_dims(X_test, axis=2)

    lstm_model = Sequential([

    lstm_model.fit(X_train_seq, y_train, epochs=10, batch_size=64, verbose=1)

    lstm_preds = lstm_model.predict(X_test_seq)

    stack_model = StackingRegressor(estimators=[
